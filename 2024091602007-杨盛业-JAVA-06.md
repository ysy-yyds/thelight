package dish;

public abstract class Dish implements Order {
    protected String name;
    protected double price;
//这里不能用private,要不然在别的类中无法使用,因此我改成了protected。
    public Dish(String name,double price) {
        this.name = name;
        this.price = price;
    }
    public void profile(){
        System.out.println("菜名"+name+price+"价格");
    }
    public abstract void cook();//子类需要用的抽象方法

    //check方法，用来生成T/F。
        public abstract boolean check();
}

package dish;

import java.util.List;//导入了类

public class system {
    private static int count = 0;//记录订单编号
    private static boolean checkResult = true;

    public void manageOrder(List<Order> dishes,Object customer) {//包含order对象列表,创建变量customer
        for (Order dish : dishes) {//增强型for循环，可以遍历列表中的每一个循环
            if(dish.check()){
                System.out.println("无法完成食物制作");
                checkResult=false;
                break;//break关键字，跳出这个循环，进入下一个循环
            }
        }
        if(checkResult) {
            count++;//编号
            System.out.println("订单号"+count);
            for (Order A : dishes){
                A.cook();
            }
    
            if(customer instanceof TableCustomer){
                TableCustomer tableCustomer = (TableCustomer) customer;
                System.out.println("顾客在在餐桌"+ tableCustomer.getTableID());//TableCustomer中的对象
            }else if (customer instanceof WechatCustomer){
                if(((WechatCustomer) customer).isTakeout()){
                    WechatCustomer wechatCustomer = (WechatCustomer) customer;
                    System.out.println("外卖地址"+wechatCustomer.getAddress());//外卖地址属性
                }else {
                    WechatCustomer wechatCustomer = (WechatCustomer) customer;
                    System.out.println("顾客是堂食，不需要额外做什么"+wechatCustomer.getTableId());
                }
    
            }
        }
    }
}
